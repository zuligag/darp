ebnf-ish grammar

Num = [0-9]+
Vsym = [A-Z][a-zA-Z0-9_]*
Verb = Vsym
  | '{' Expr ( ';' Expr )*  '}'
  | VSym ':' Verb
Noun = Num
  | [a-z][a-zA-Z0-9_]*
  | '[' Noun+ ']'
  | '(' ',' Noun* ')'
